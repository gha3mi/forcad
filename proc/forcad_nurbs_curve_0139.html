    subroutine forcad_nurbs_curve_0139(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: curve_fd
        real(rk) :: h, u0, um, up, knot_fd(6)
        real(rk), allocatable :: Xc_fd(:,:), Wc_fd(:)
        real(rk), allocatable :: T0(:), Tm(:), Tp(:), dT0(:), dTm(:), dTp(:), d2T0(:), d2Tm(:), d2Tp(:)

        allocate(Xc_fd(3, 3), Wc_fd(3))
        Xc_fd(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xc_fd(2,:) = [0.0_rk, 5.0_rk, 0.0_rk]
        Xc_fd(3,:) = [5.0_rk, 5.0_rk, 0.0_rk]
        Wc_fd = [1.0_rk, 1.1_rk, 1.0_rk]
        knot_fd = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        h = 1.0e-6_rk
        u0 = 0.5_rk
        um = u0 - h
        up = u0 + h

        call curve_fd%set(knot_fd, Xc_fd, Wc_fd)
        call curve_fd%derivative2(Xt=u0, d2Tgc=d2T0, dTgc=dT0, Tgc=T0)
        call curve_fd%derivative2(Xt=um, d2Tgc=d2Tm, dTgc=dTm, Tgc=Tm)
        call curve_fd%derivative2(Xt=up, d2Tgc=d2Tp, dTgc=dTp, Tgc=Tp)

        call curve_fd%finalize()
        call curve_fd%set(knot_fd, Xc_fd)
        call curve_fd%derivative2(Xt=u0, d2Tgc=d2T0, dTgc=dT0, Tgc=T0)
        call curve_fd%derivative2(Xt=um, d2Tgc=d2Tm, dTgc=dTm, Tgc=Tm)
        call curve_fd%derivative2(Xt=up, d2Tgc=d2Tp, dTgc=dTp, Tgc=Tp)

        call ut%test(ti)%check(&
            name     = "derivative2(): bspline central d2T",&
            res      = norm2((dTp - dTm)/(2.0_rk*h) - d2T0),&
            expected = 0.0_rk,&
            tol      = 1.0e-7_rk,&
            msg      = "Derivative2(): bspline central d2T is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0139
