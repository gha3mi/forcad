    subroutine forcad_nurbs_curve_0130(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: nurbs_curve_tol = 1.0e-5_rk
        type(nurbs_curve) :: bounded_curve
        real(rk), allocatable :: Xg_out(:), Tgc_out(:)

        call bounded_curve%set_circle([0.0_rk, 0.0_rk, 0.0_rk], 1.0_rk)
        Xg_out = bounded_curve%cmp_Xg(-0.25_rk)
        call bounded_curve%basis(-0.25_rk, Tgc_out)

        call ut%test(ti)%check(&
            name     = "cmp_Xg(): rational outside active domain",&
            res      = all(abs(Xg_out) <= nurbs_curve_tol),&
            expected = .true.,&
            msg      = "Out-of-domain rational curve evaluation should be zero, not NaN",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0130
