    subroutine forcad_nurbs_curve_0178(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: cempty
        integer, allocatable :: ivals(:)
        real(rk), allocatable :: rvals(:), rmat(:,:)

        ivals = cempty%get_multiplicity()

        ivals = cempty%get_continuity()

        rmat = cempty%get_Xc()
        rvals = cempty%get_Xc(1)
        rmat = cempty%get_Xg()
        rvals = cempty%get_Xg(1)
        rvals = cempty%get_Wc()
        rvals = cempty%get_Xt()
        rvals = cempty%get_knot()
        rvals = cempty%cmp_Xg(0.0_rk)

        call ut%test(ti)%check(&
            name     = "cmp_Xg(): unset curve returns empty",&
            res      = size(rvals),&
            expected = 0,&
            msg      = "Unset curve point evaluation should not terminate",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0178
