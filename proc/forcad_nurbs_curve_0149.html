    subroutine forcad_nurbs_curve_0149(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: sh0
        type(nurbs_curve) :: shr
        type(nurbs_curve) :: shb
        real(rk), allocatable :: Xc_s(:,:)
        real(rk), allocatable :: Wc_s(:)
        real(rk), allocatable :: Xc0(:,:)
        real(rk), allocatable :: S1(:,:)
        real(rk), allocatable :: S2(:,:)
        real(rk), allocatable :: Bdirect(:,:)
        real(rk), allocatable :: knot_s(:), u(:)
        integer, allocatable :: r(:)
        integer :: ncoord
        integer :: nc0
        integer :: ndof_old

        allocate(Xc_s(2,3), Wc_s(2), knot_s(4))
        Xc_s(1,:) = [-2.5_rk, 0.0_rk, 0.0_rk]
        Xc_s(2,:) = [ 2.5_rk, 0.0_rk, 0.0_rk]
        Wc_s = [1.0_rk, 0.5_rk]
        knot_s = [0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk]

        call shr%set(knot_s, Xc_s, Wc_s)
        sh0 = shr
        Xc0 = sh0%get_Xc()
        ncoord = size(Xc0, 2)
        nc0 = size(Xc0, 1)
        ndof_old = nc0*ncoord

        call shr%elevate_degree(3, Bs=S1)
        call shb%set(sh0%get_knot(), Xc0, sh0%get_Wc())
        call shb%elevate_degree(3, B=Bdirect)
        u = linspace(0.0_rk, 1.0_rk, 7)
        u = u(2:6)
        allocate(r(size(u)), source=2)
        call shr%insert_knots(u, r, Bs=S2)
        call shb%insert_knots(u, r, B=Bdirect)

        call ut%test(ti)%check(&
            name     = "insert_knots(): curve B-only map",&
            res      = maxval(abs(Bdirect - kron_eye(S2, ncoord))),&
            expected = 0.0_rk,&
            tol      = 1.0e-10_rk,&
            msg      = "Insert_knots(): curve B-only map is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0149
