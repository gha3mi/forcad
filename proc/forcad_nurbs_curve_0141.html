    subroutine forcad_nurbs_curve_0141(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: curve_ref
        real(rk), allocatable :: Xc_ref(:,:)
        real(rk), allocatable :: Xg_ref(:,:)
        real(rk) :: knot_ref(7), Xt_ref(5)
        integer :: ii

        allocate(Xc_ref(4, 3), source=0.0_rk)
        do ii = 1, 4
            Xc_ref(ii, 1) = real(ii - 1, rk)
            Xc_ref(ii, 2) = sin(real(ii - 1, rk))
        end do
        knot_ref = [0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 5.0_rk, 6.0_rk]
        Xt_ref = [2.0_rk, 2.25_rk, 2.75_rk, 3.5_rk, 4.0_rk]

        call curve_ref%set(knot_ref, Xc_ref)
        call curve_ref%create(Xt=Xt_ref)
        Xg_ref = curve_ref%get_Xg()

        call curve_ref%insert_knots([2.5_rk], [1])
        call curve_ref%create(Xt=Xt_ref)

        call curve_ref%remove_knots([2.5_rk], [1])
        call curve_ref%create(Xt=Xt_ref)

        call ut%test(ti)%check(&
            name     = "remove_knots(): non-open curve geometry",&
            res      = maxval(abs(curve_ref%get_Xg() - Xg_ref)),&
            expected = 0.0_rk,&
            tol      = 1.0e-10_rk,&
            msg      = "Non-open curve knot removal should preserve geometry",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0141
