    subroutine forcad_nurbs_curve_0147(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: curve_ref
        real(rk), parameter :: knot_ref(12) = [&
            0.0_rk, 0.0_rk, 0.0_rk, 0.0_rk, 0.18_rk, 0.42_rk,&
            0.73_rk, 0.88_rk, 1.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        real(rk), parameter :: Xc_ref(8,3) = reshape([&
            0.0_rk, 0.8_rk, 1.9_rk, 3.0_rk, 4.1_rk, 5.0_rk, 5.8_rk, 6.6_rk,&
            0.0_rk, 1.2_rk, 0.7_rk,-0.8_rk,-1.4_rk,-0.2_rk, 1.1_rk, 0.4_rk,&
            0.0_rk, 0.4_rk, 1.5_rk, 2.2_rk, 1.3_rk, 0.1_rk, 0.8_rk, 2.0_rk], [8,3])
        real(rk), parameter :: Wc_ref(8) = [&
            1.0_rk, 0.82_rk, 1.25_rk, 0.94_rk, 1.18_rk, 0.88_rk, 1.12_rk, 1.0_rk]
        real(rk), parameter :: Xt_ref(9) = [&
            0.0_rk, 0.07_rk, 0.20_rk, 0.35_rk, 0.50_rk, 0.67_rk, 0.82_rk, 0.95_rk, 1.0_rk]
        real(rk), allocatable :: Xg_ref(:,:)

        call curve_ref%set(&
            knot   = knot_ref,&
            Xc     = Xc_ref,&
            Wc     = Wc_ref,&
            degree = 3)
        call curve_ref%create(Xt = Xt_ref)
        Xg_ref = curve_ref%get_Xg()
        call curve_ref%insert_knots(&
            Xth = [0.30_rk, 0.60_rk],&
            r   = [1, 1])
        call curve_ref%elevate_degree(t = 1)

        call curve_ref%create(Xt = Xt_ref)

        call ut%test(ti)%check(&
            name     = "elevate_degree(): inserted rational geometry",&
            res      = maxval(abs(curve_ref%get_Xg()-Xg_ref)),&
            expected = 0.0_rk,&
            tol      = 1.0e-10_rk,&
            msg      = "Elevate_degree(): inserted rational geometry is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0147
