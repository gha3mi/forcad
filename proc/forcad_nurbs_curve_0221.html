    subroutine forcad_nurbs_curve_0221(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: full_break_curve
        real(rk) :: full_break_knot(9)
        real(rk) :: full_break_Xc(6,3)
        real(rk) :: full_break_Wc(6)
        real(rk) :: boundary_Xc(3,3)
        real(rk) :: full_break_Xt(9)
        real(rk), allocatable :: full_break_Xg(:,:)
        real(rk), allocatable :: full_break_reference(:,:)

        full_break_knot = [0.0_rk, 0.0_rk, 0.0_rk, 0.5_rk, 0.5_rk, 0.5_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        full_break_Xc = reshape([&
            0.0_rk, 0.0_rk, 0.0_rk,&
            0.2_rk, 0.8_rk, 0.1_rk,&
            0.5_rk, 0.0_rk, 0.2_rk,&
            0.5_rk, 2.0_rk, 0.3_rk,&
            0.8_rk, 2.8_rk, 0.4_rk,&
            1.0_rk, 2.0_rk, 0.5_rk], shape(full_break_Xc), order=[2,1])
        boundary_Xc = full_break_Xc(1:3,:)
        full_break_Wc = [1.0_rk, 0.7_rk, 1.2_rk, 0.9_rk, 1.3_rk, 1.0_rk]
        full_break_Xt = [0.0_rk, 0.125_rk, 0.25_rk, 0.375_rk, 0.5_rk, &
            0.625_rk, 0.75_rk, 0.875_rk, 1.0_rk]

        call full_break_curve%set(full_break_knot, full_break_Xc, full_break_Wc, degree=2)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_reference = full_break_curve%get_Xg()
        call full_break_curve%elevate_degree(1)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_Xg = full_break_curve%get_Xg()

        call ut%test(ti)%check(&
            name     = "full-break elevation geometry",&
            res      = full_break_Xg,&
            expected = full_break_reference,&
            tol      = 5.0e-12_rk,&
            msg      = "Full-break elevation geometry is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0221
