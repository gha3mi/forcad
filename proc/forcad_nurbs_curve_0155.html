    subroutine forcad_nurbs_curve_0155(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: cbad
        real(rk) :: Xbad(2,3)
        real(rk) :: Xnew(3,3)
        real(rk) :: Wbad(2)
        real(rk) :: Wbad3(3)
        real(rk) :: Wgood(2)

        Xbad(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xbad(2,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        Xnew(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xnew(2,:) = [1.0_rk, 0.5_rk, 0.0_rk]
        Xnew(3,:) = [2.0_rk, 0.0_rk, 0.0_rk]
        Wbad = [1.0_rk, 0.0_rk]
        Wbad3 = [1.0_rk, 0.0_rk, 1.0_rk]
        Wgood = 1.0_rk
        call cbad%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wbad)

        call ut%test(ti)%check(&
            name     = "is_rational(): invalid curve state",&
            res      = cbad%is_rational(),&
            expected = .false.,&
            msg      = "Curve is_rational should return false for invalid objects",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0155
