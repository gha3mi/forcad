    subroutine forcad_nurbs_curve_0241(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: PERIODIC_TOL = 4096.0_rk*epsilon(1.0_rk)
        type(nurbs_curve) :: periodic_curve
        real(rk) :: periodic_knot(9), periodic_Xc(6,2), periodic_Wc(6)
        real(rk), allocatable :: T_start(:), T_end(:), dT_start(:), dT_end(:)

        periodic_knot = [-2.2_rk, -1.6_rk, 0.0_rk, 0.7_rk, 1.8_rk, 2.4_rk, 4.0_rk, 4.7_rk, 5.8_rk]
        periodic_Xc(1,:) = [ 1.0_rk, 0.0_rk]
        periodic_Xc(2,:) = [ 0.0_rk, 1.0_rk]
        periodic_Xc(3,:) = [-1.0_rk, 0.0_rk]
        periodic_Xc(4,:) = [ 0.0_rk,-1.0_rk]
        periodic_Xc(5,:) = periodic_Xc(1,:)
        periodic_Xc(6,:) = periodic_Xc(2,:)

        call periodic_curve%set(periodic_knot, periodic_Xc, degree=2, wrap_parameters=.true.)
        call compute_dTgc(0.0_rk, periodic_knot, 2, 6, dT_start, T_start)
        call compute_dTgc(4.0_rk, periodic_knot, 2, 6, dT_end, T_end)
        call periodic_curve%derivative(2.75_rk, dT_start, T_start)
        call periodic_curve%derivative(-1.25_rk, dT_end, T_end)

        periodic_Wc = [1.0_rk, 1.4_rk, 0.8_rk, 1.2_rk, 1.0_rk, 1.4_rk]
        call periodic_curve%finalize()
        call periodic_curve%set(&
            knot           = periodic_knot,&
            Xc             = periodic_Xc,&
            Wc             = periodic_Wc,&
            degree         = 2,&
            wrap_parameters = .true.)
        call compute_dTgc(0.0_rk, periodic_knot, 2, 6, periodic_Wc, dT_start, T_start)
        call compute_dTgc(4.0_rk, periodic_knot, 2, 6, periodic_Wc, dT_end, T_end)

        call ut%test(ti)%check(&
            name     = "wrapped periodic NURBS geometry",&
            res      = periodic_curve%cmp_Xg(10.75_rk),&
            expected = periodic_curve%cmp_Xg(2.75_rk),&
            tol      = PERIODIC_TOL,&
            msg      = "Wrapped periodic NURBS geometry is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0241
