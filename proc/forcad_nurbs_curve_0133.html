    subroutine forcad_nurbs_curve_0133(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: curve_ref, curve_scaled
        real(rk) :: knot_scale(6), u
        real(rk), allocatable :: Xc_scale(:,:), W_scale(:)
        real(rk), allocatable :: Xg_ref(:), Xg_scaled(:), T_ref(:), T_scaled(:), dT_ref(:), dT_scaled(:)

        allocate(Xc_scale(3, 3), W_scale(3))
        Xc_scale(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xc_scale(2,:) = [1.0_rk, 2.0_rk, 0.0_rk]
        Xc_scale(3,:) = [3.0_rk, 1.0_rk, 0.0_rk]
        W_scale = [1.0_rk, 0.7_rk, 1.3_rk]
        knot_scale = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        u = 0.37_rk

        call curve_ref%set(knot_scale, Xc_scale, W_scale)
        call curve_scaled%set(knot_scale, Xc_scale, 1.0e-20_rk*W_scale)
        Xg_ref = curve_ref%cmp_Xg(u)
        Xg_scaled = curve_scaled%cmp_Xg(u)
        call curve_ref%derivative(u, dT_ref, T_ref)
        call curve_scaled%derivative(u, dT_scaled, T_scaled)

        call ut%test(ti)%check(&
            name     = "rational curve weight-scale Xg",&
            res      = maxval(abs(Xg_ref - Xg_scaled)),&
            expected = 0.0_rk,&
            tol      = 1.0e-10_rk,&
            msg      = "Rational curve weight-scale Xg is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0133
