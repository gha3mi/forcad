    subroutine forcad_nurbs_curve_0162(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: cinsert
        type(nurbs_curve) :: cremove
        real(rk) :: knot_bad(6), Xc_bad(3,3), Wc_bad(3)

        knot_bad = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        Xc_bad = 0.0_rk
        Xc_bad(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xc_bad(2,:) = [0.5_rk, 1.0_rk, 0.0_rk]
        Xc_bad(3,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        Wc_bad = [1.0_rk, 0.8_rk, 1.2_rk]

        call cinsert%set(knot_bad, Xc_bad)
        call cinsert%insert_knots([0.5_rk], [4])

        call cremove%set(knot_bad, Xc_bad)
        call cremove%remove_knots([0.5_rk], [1])

        call ut%test(ti)%check(&
            name     = "curve bad remove",&
            res      = cremove%err%ok,&
            expected = .false.,&
            msg      = "curve bad remove",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0162
