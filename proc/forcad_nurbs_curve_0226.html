    subroutine forcad_nurbs_curve_0226(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: full_break_curve
        real(rk) :: full_break_knot(9), nonopen_knot(9), full_break_Xc(6,3), full_break_Wc(6)
        real(rk) :: boundary_Xc(3,3), removable_Xc(6,3), removable_Wc(6), full_break_Xt(9)
        real(rk), allocatable :: full_break_Xg(:,:), full_break_reference(:,:), boundary_knot(:)

        full_break_knot = [0.0_rk, 0.0_rk, 0.0_rk, 0.5_rk, 0.5_rk, 0.5_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        full_break_Xc = reshape([&
            0.0_rk, 0.0_rk, 0.0_rk,&
            0.2_rk, 0.8_rk, 0.1_rk,&
            0.5_rk, 0.0_rk, 0.2_rk,&
            0.5_rk, 2.0_rk, 0.3_rk,&
            0.8_rk, 2.8_rk, 0.4_rk,&
            1.0_rk, 2.0_rk, 0.5_rk], shape(full_break_Xc), order=[2,1])
        boundary_Xc = full_break_Xc(1:3,:)
        full_break_Wc = [1.0_rk, 0.7_rk, 1.2_rk, 0.9_rk, 1.3_rk, 1.0_rk]
        full_break_Xt = [0.0_rk, 0.125_rk, 0.25_rk, 0.375_rk, 0.5_rk, &
            0.625_rk, 0.75_rk, 0.875_rk, 1.0_rk]

        call full_break_curve%set(full_break_knot, full_break_Xc, full_break_Wc, degree=2)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_reference = full_break_curve%get_Xg()
        call full_break_curve%elevate_degree(1)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_Xg = full_break_curve%get_Xg()

        nonopen_knot = [-2.0_rk, -1.0_rk, 0.0_rk, 0.5_rk, 0.5_rk, 0.5_rk, 1.0_rk, 2.0_rk, 3.0_rk]
        call full_break_curve%finalize()
        call full_break_curve%set(nonopen_knot, full_break_Xc, full_break_Wc, degree=2)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_reference = full_break_curve%get_Xg()
        call full_break_curve%elevate_degree(1)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_Xg = full_break_curve%get_Xg()

        removable_Xc = full_break_Xc
        removable_Wc = full_break_Wc
        removable_Xc(4,:) = removable_Xc(3,:)
        removable_Wc(4) = removable_Wc(3)
        call full_break_curve%finalize()
        call full_break_curve%set(&
            knot   = full_break_knot,&
            Xc     = removable_Xc,&
            Wc     = removable_Wc,&
            degree = 2)
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_reference = full_break_curve%get_Xg()
        call full_break_curve%remove_knots(&
            Xth = [0.5_rk],&
            r   = [1])
        call full_break_curve%create(Xt=full_break_Xt)
        full_break_Xg = full_break_curve%get_Xg()
        boundary_knot = full_break_curve%get_knot()

        call ut%test(ti)%check(&
            name     = "full-break removal geometry",&
            res      = full_break_Xg,&
            expected = full_break_reference,&
            tol      = 5.0e-12_rk,&
            msg      = "Removing a redundant full-break knot must preserve geometry",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0226
