    subroutine forcad_nurbs_curve_0253(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: rank_one_curve, copied_curve
        real(rk) :: rank_one_knot(6), rank_one_Xc(3), rank_one_Wc(3), expected_Xc(3,3)

        rank_one_knot = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        rank_one_Xc = [0.0_rk, 1.0_rk, 2.0_rk]
        rank_one_Wc = [1.0_rk, 0.75_rk, 1.0_rk]
        expected_Xc = 0.0_rk
        expected_Xc(:,1) = rank_one_Xc

        call rank_one_curve%set(rank_one_knot, rank_one_Xc, rank_one_Wc)

        copied_curve = rank_one_curve
        call rank_one_curve%modify_Xc(5.0_rk, 1, 1)

        call ut%test(ti)%check(&
            name     = "curve intrinsic assignment deep copy",&
            res      = copied_curve%get_Xc(1,1),&
            expected = 0.0_rk,&
            tol      = 0.0_rk,&
            msg      = "Curve intrinsic assignment deep copy is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0253
