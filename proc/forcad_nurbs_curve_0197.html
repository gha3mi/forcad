    subroutine forcad_nurbs_curve_0197(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: HESSIAN_TOL = 4096.0_rk*epsilon(1.0_rk)
        type(nurbs_curve) :: curve
        real(rk) :: hessian_knot(6)
        real(rk) :: hessian_Xc(3,2)
        real(rk) :: dL
        real(rk), allocatable :: T(:), dT_dX(:,:), d2T_dX2(:)

        hessian_knot = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        hessian_Xc(1,:) = [0.0_rk, 0.0_rk]
        hessian_Xc(2,:) = [0.5_rk, 0.0_rk]
        hessian_Xc(3,:) = [1.0_rk, 1.0_rk]

        call curve%set(&
            knot   = hessian_knot,&
            Xc     = hessian_Xc,&
            degree = 2)
        call curve%ansatz(&
            ie          = 1,&
            ig          = 1,&
            Tgc         = T,&
            dTgc_dXg    = dT_dX,&
            d2Tgc_dXg2  = d2T_dX2,&
            dL          = dL,&
            ngauss      = 1)

        call ut%test(ti)%check(&
            name     = "ansatz y chain rule",&
            res      = sum(d2T_dX2*hessian_Xc(:,2)),&
            expected = 0.5_rk,&
            tol      = HESSIAN_TOL,&
            msg      = "Ansatz y chain rule is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0197
