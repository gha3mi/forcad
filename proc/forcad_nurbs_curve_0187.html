    subroutine forcad_nurbs_curve_0187(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: rational_order
        type(nurbs_curve) :: active_curve
        real(rk) :: Xc_order(2,3), Wc_order(2), expected_order(2), magnitude
        real(rk) :: Xc_active(5,3), Wc_active(5)
        real(rk), allocatable :: Dgc_order(:), Dgc_order_elem(:), Dgc_order_vector(:,:)
        real(rk), allocatable :: Dgc_active(:,:)
        real(rk), allocatable :: Dgc_dense(:,:)
        real(rk), allocatable :: Dgc_reconstructed(:,:)
        real(rk), allocatable :: Dgc_active_scalar(:)
        integer, allocatable :: first_active(:)
        integer :: first_scalar, point, local

        Xc_order = 0.0_rk
        Wc_order = [1.0_rk, 2.0_rk]
        call rational_order%set(&
            knot = [0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk],&
            Xc   = Xc_order,&
            Wc   = Wc_order)
        magnitude = 48.0_rk/(1.25_rk**5)
        expected_order = [magnitude,-magnitude]

        call rational_order%derivative_order(&
            Xt    = 0.25_rk,&
            order = 4,&
            Dgc   = Dgc_order)

        call rational_order%derivative_order(&
            Xt    = 0.25_rk,&
            order = 4,&
            Dgc   = Dgc_order_elem,&
            elem  = [2,1,0])

        call rational_order%derivative_order(&
            order = 4,&
            Dgc   = Dgc_order_vector,&
            Xt    = [0.25_rk,0.5_rk])

        call rational_order%derivative_order_active(&
            Xt           = 0.25_rk,&
            order        = 4,&
            first_active = first_scalar,&
            Dgc          = Dgc_active_scalar)

        Xc_active = 0.0_rk
        Wc_active = [1.0_rk,1.2_rk,0.7_rk,1.5_rk,0.9_rk]
        call active_curve%set(&
            knot = [0.0_rk,0.0_rk,0.0_rk,0.2_rk,0.65_rk,1.0_rk,1.0_rk,1.0_rk],&
            Xc   = Xc_active,&
            Wc   = Wc_active)
        call active_curve%derivative_order(&
            order = 2,&
            Dgc   = Dgc_dense,&
            Xt    = [0.1_rk,0.4_rk,0.9_rk])
        call active_curve%derivative_order_active(&
            Xt           = [0.1_rk,0.4_rk,0.9_rk],&
            order        = 2,&
            first_active = first_active,&
            Dgc          = Dgc_active)
        allocate(Dgc_reconstructed(size(Dgc_dense,1),size(Dgc_dense,2)), source=0.0_rk)
        do point = 1, size(first_active)
            do local = 0, size(Dgc_active,1)-1
                Dgc_reconstructed(point,first_active(point)+local) = Dgc_active(local+1,point)
            end do
        end do

        call ut%test(ti)%check(&
            name     = "derivative_order_active(): dense reconstruction",&
            res      = Dgc_reconstructed,&
            expected = Dgc_dense,&
            tol      = 128.0_rk*epsilon(1.0_rk)*max(1.0_rk,maxval(abs(Dgc_dense))),&
            msg      = "Derivative_order_active(): dense reconstruction is incorrect.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0187
