    subroutine forcad_nurbs_curve_0129(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: nurbs_curve_tol = 1.0e-5_rk
        type(nurbs_curve) :: repeated
        real(rk), allocatable :: Xc_repeated(:,:), Tgc_repeated(:), Xt_repeated(:)
        real(rk) :: knot_repeated(9)
        integer :: ii

        allocate(Xc_repeated(6, 3), source=0.0_rk)
        do ii = 1, 6
            Xc_repeated(ii,1) = real(ii - 1, rk)
            Xc_repeated(ii,2) = 0.25_rk*real((ii - 1)**2, rk)
        end do
        knot_repeated = [0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 3.0_rk, 3.0_rk, 4.0_rk, 6.0_rk, 9.0_rk]

        call repeated%set(knot_repeated, Xc_repeated, degree=2)
        call repeated%create(res=5)
        Xt_repeated = repeated%get_Xt()
        call repeated%basis(3.25_rk, Tgc_repeated)

        call ut%test(ti)%check(&
            name     = "set(): repeated nonuniform non-open curve knot vector",&
            res      = merge(1, 0, repeated%err%ok .and. repeated%get_degree() == 2 .and. repeated%get_nc() == 6 .and. &
                abs(Xt_repeated(1) - 2.0_rk) <= nurbs_curve_tol .and. abs(Xt_repeated(size(Xt_repeated)) - 4.0_rk) <= nurbs_curve_tol .and. &
                abs(sum(Tgc_repeated) - 1.0_rk) <= nurbs_curve_tol),&
            expected = 1,&
            msg      = "Curve should support repeated, nonuniform, non-open knot vectors",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0129
