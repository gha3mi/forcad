    subroutine forcad_nurbs_curve_0068(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: nurbs_curve_tol = 1.0e-5_rk
        character(len=*), parameter :: nurbs_curve_fXc_nurbs = "vtk/test_nurbs_curve_Xc.vtk"
        character(len=*), parameter :: nurbs_curve_fXg_nurbs = "vtk/test_nurbs_curve_Xg.vtk"
        character(len=*), parameter :: nurbs_curve_fXth_nurbs = "vtk/test_nurbs_curve_Xth.vtk"
        character(len=*), parameter :: nurbs_curve_fXthXg_nurbs = "vtk/test_nurbs_curve_Xth_in_Xg.vtk"
        character(len=*), parameter :: nurbs_curve_fIgs_nurbs = "iges/test_nurbs_curve.iges"
        character(len=*), parameter :: nurbs_curve_fXc_bsp = "vtk/test_bsp_curve_Xc.vtk"
        character(len=*), parameter :: nurbs_curve_fXg_bsp = "vtk/test_bsp_curve_Xg.vtk"
        character(len=*), parameter :: nurbs_curve_fXth_bsp = "vtk/test_bsp_curve_Xth.vtk"
        character(len=*), parameter :: nurbs_curve_fXthXg_bsp = "vtk/test_bsp_curve_Xth_in_Xg.vtk"
        character(len=*), parameter :: nurbs_curve_fIgs_bsp = "iges/test_bsp_curve.iges"
        type(nurbs_curve) :: nurbs, bsp
        real(rk), allocatable :: Xc(:,:), Wc(:), Xg(:,:), Xgb(:,:)
        integer, allocatable :: elemConn(:,:)
        real(rk) :: knot(6), nearest_Xg(3), nearest_Xt, length, lengthb
        integer :: id
        logical :: okXc_nurbs, okXg_nurbs, okXth_nurbs, okXthXg_nurbs, okIges_nurbs
        logical :: okXc_bsp, okXg_bsp, okXth_bsp, okXthXg_bsp, okIges_bsp

        allocate(Xc(3, 3))
        Xc(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xc(2,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        Xc(3,:) = [2.0_rk, 0.0_rk, 0.0_rk]
        allocate(Wc(3))
        Wc = [1.0_rk, 0.9_rk, 0.8_rk]
        knot = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]

        call nurbs%set(knot, Xc, Wc)
        call bsp%set(knot, Xc)
        call nurbs%set(degree=nurbs%get_degree(), nc=nurbs%get_nc(), Xc=nurbs%get_Xc(), Wc=nurbs%get_Wc())
        call bsp%set(degree=bsp%get_degree(), nc=bsp%get_nc(), Xc=bsp%get_Xc())
        call nurbs%create(res=23)
        call bsp%create(res=23)

        call nurbs%export_Xc(nurbs_curve_fXc_nurbs)
        call bsp%export_Xc(nurbs_curve_fXc_bsp)
        call nurbs%export_Xg(nurbs_curve_fXg_nurbs)
        call bsp%export_Xg(nurbs_curve_fXg_bsp)
        call nurbs%export_Xth(nurbs_curve_fXth_nurbs)
        call bsp%export_Xth(nurbs_curve_fXth_bsp)
        call nurbs%export_Xth_in_Xg(nurbs_curve_fXthXg_nurbs, res=7)
        call bsp%export_Xth_in_Xg(nurbs_curve_fXthXg_bsp, res=7)
        call nurbs%export_iges(nurbs_curve_fIgs_nurbs)
        call bsp%export_iges(nurbs_curve_fIgs_bsp)
        inquire(file=nurbs_curve_fXc_nurbs, exist=okXc_nurbs)
        inquire(file=nurbs_curve_fXg_nurbs, exist=okXg_nurbs)
        inquire(file=nurbs_curve_fXth_nurbs, exist=okXth_nurbs)
        inquire(file=nurbs_curve_fXthXg_nurbs, exist=okXthXg_nurbs)
        inquire(file=nurbs_curve_fIgs_nurbs, exist=okIges_nurbs)
        inquire(file=nurbs_curve_fXc_bsp, exist=okXc_bsp)
        inquire(file=nurbs_curve_fXg_bsp, exist=okXg_bsp)
        inquire(file=nurbs_curve_fXth_bsp, exist=okXth_bsp)
        inquire(file=nurbs_curve_fXthXg_bsp, exist=okXthXg_bsp)
        inquire(file=nurbs_curve_fIgs_bsp, exist=okIges_bsp)

        call nurbs%cmp_length(length)
        call bsp%cmp_length(lengthb)

        call nurbs%nearest_point([0.0_rk, 0.0_rk, 0.5_rk], nearest_Xg, nearest_Xt, id)

        call bsp%nearest_point([0.0_rk, 0.0_rk, 0.5_rk], nearest_Xg, nearest_Xt, id)

        call nurbs%nearest_point2([0.0_rk, 0.0_rk, 0.5_rk], 1.0e-10_rk, 10, nearest_Xt, nearest_Xg)

        call bsp%nearest_point2([0.0_rk, 0.0_rk, 0.5_rk], 1.0e-10_rk, 10, nearest_Xt, nearest_Xg)

        Xg = nurbs%get_Xg()
        Xgb = bsp%get_Xg()

        call nurbs%set([0.0_rk, 1.0_rk], 2, [-1, -1], Xc, Wc)
        call bsp%set([0.0_rk, 1.0_rk], 2, [-1, -1], Xc)
        call nurbs%create(res=23)
        call bsp%create(res=23)

        call nurbs%set(Xc, Wc)
        call bsp%set(Xc)
        call nurbs%create(res=23)
        call bsp%create(res=23)

        call nurbs%create(Xt=nurbs%get_Xt())
        call bsp%create(Xt=bsp%get_Xt())

        call nurbs%export_Xc(nurbs_curve_fXc_nurbs)
        call bsp%export_Xc(nurbs_curve_fXc_bsp)
        call nurbs%export_Xg(nurbs_curve_fXg_nurbs)
        call bsp%export_Xg(nurbs_curve_fXg_bsp)
        call nurbs%export_Xth(nurbs_curve_fXth_nurbs)
        call bsp%export_Xth(nurbs_curve_fXth_bsp)
        call nurbs%export_iges(nurbs_curve_fIgs_nurbs)
        call bsp%export_iges(nurbs_curve_fIgs_bsp)
        inquire(file=nurbs_curve_fXc_nurbs, exist=okXc_nurbs)
        inquire(file=nurbs_curve_fXg_nurbs, exist=okXg_nurbs)
        inquire(file=nurbs_curve_fXth_nurbs, exist=okXth_nurbs)
        inquire(file=nurbs_curve_fIgs_nurbs, exist=okIges_nurbs)
        inquire(file=nurbs_curve_fXc_bsp, exist=okXc_bsp)
        inquire(file=nurbs_curve_fXg_bsp, exist=okXg_bsp)
        inquire(file=nurbs_curve_fXth_bsp, exist=okXth_bsp)
        inquire(file=nurbs_curve_fIgs_bsp, exist=okIges_bsp)

        call nurbs%cmp_nc()
        call bsp%cmp_nc()

        elemConn = nurbs%cmp_elem_Xc_vis(2)
        call nurbs%set_elem_Xc_vis(elemConn)
        deallocate(elemConn)
        elemConn = nurbs%cmp_elem_Xc_vis()
        call nurbs%set_elem_Xc_vis(elemConn)
        deallocate(elemConn)
        elemConn = bsp%cmp_elem_Xc_vis(2)
        call bsp%set_elem_Xc_vis(elemConn)
        deallocate(elemConn)
        elemConn = bsp%cmp_elem_Xc_vis()
        call bsp%set_elem_Xc_vis(elemConn)
        deallocate(elemConn)
        elemConn = nurbs%cmp_elem_Xg_vis(2)
        call nurbs%set_elem_Xg_vis(elemConn)
        deallocate(elemConn)
        elemConn = nurbs%cmp_elem_Xg_vis()
        call nurbs%set_elem_Xg_vis(elemConn)
        deallocate(elemConn)
        elemConn = bsp%cmp_elem_Xg_vis(2)
        call bsp%set_elem_Xg_vis(elemConn)
        deallocate(elemConn)
        elemConn = bsp%cmp_elem_Xg_vis()
        call bsp%set_elem_Xg_vis(elemConn)
        deallocate(elemConn)
        elemConn = nurbs%cmp_elem()
        call nurbs%set_elem(elemConn)
        deallocate(elemConn)
        elemConn = bsp%cmp_elem()
        call bsp%set_elem(elemConn)
        deallocate(elemConn)

        call nurbs%modify_Xc(Xc(1,1), 1, 1)
        call bsp%modify_Xc(Xc(1,1), 1, 1)
        call nurbs%modify_Wc(Wc(1), 1)
        call nurbs%create()
        call bsp%create()
        call nurbs%export_Xc(nurbs_curve_fXc_nurbs)
        call bsp%export_Xc(nurbs_curve_fXc_bsp)
        call nurbs%export_Xg(nurbs_curve_fXg_nurbs)
        call bsp%export_Xg(nurbs_curve_fXg_bsp)

        call ut%test(ti)%check(&
            name     = "modify_Xc: geometry preserved (B-spline)",&
            res      = bsp%get_Xg(),&
            expected = Xgb,&
            tol      = nurbs_curve_tol,&
            msg      = "B-spline geometry changed after modify_Xc",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0068
