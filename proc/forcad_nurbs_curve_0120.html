    subroutine forcad_nurbs_curve_0120(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: singular_fit
        real(rk), parameter :: singular_knot(6) = [0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk]
        real(rk), parameter :: singular_Xc(3,3) = reshape([0.0_rk, 1.0_rk, 2.0_rk, &
            0.0_rk, 0.0_rk, 0.0_rk, 0.0_rk, 0.0_rk, 0.0_rk], [3,3])
        real(rk) :: Xt_bad(3), Xdata_bad(3, 3)

        Xt_bad = 0.0_rk
        Xdata_bad = 0.0_rk
        call singular_fit%set(singular_knot, singular_Xc)
        call singular_fit%lsq_fit_bspline(Xt_bad, Xdata_bad, 3)

        call ut%test(ti)%check(&
            name     = "lsq_fit_bspline(): singular banded solve diagnostic",&
            res      = singular_fit%err%ok,&
            expected = .false.,&
            msg      = "Singular B-spline least-squares fit should be recoverable",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0120
