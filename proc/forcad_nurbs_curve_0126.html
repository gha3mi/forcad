    subroutine forcad_nurbs_curve_0126(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: nurbs_curve_tol = 1.0e-5_rk
        type(nurbs_curve) :: nonopen
        real(rk), allocatable :: Xc_nonopen(:,:), Xt_nonopen(:), Tgc_nonopen(:), dTgc_nonopen(:)
        real(rk) :: knot_nonopen(7)
        integer :: ii

        allocate(Xc_nonopen(4, 3), source=0.0_rk)
        do ii = 1, 4
            Xc_nonopen(ii,1) = real(ii - 1, rk)
        end do
        knot_nonopen = [0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 5.0_rk, 6.0_rk]

        call nonopen%set(knot_nonopen, Xc_nonopen)
        call nonopen%create(res=5)
        Xt_nonopen = nonopen%get_Xt()

        call nonopen%basis(2.5_rk, Tgc_nonopen)

        call nonopen%basis(1.5_rk, Tgc_nonopen)

        call nonopen%derivative(2.5_rk, dTgc_nonopen, Tgc_nonopen, elem=[0, 5])

        call ut%test(ti)%check(&
            name     = "derivative(): non-open invalid local element entries are zero",&
            res      = max(maxval(abs(Tgc_nonopen)), maxval(abs(dTgc_nonopen))),&
            expected = 0.0_rk,&
            tol      = nurbs_curve_tol,&
            msg      = "Derivative(): non-open invalid local element entries are zero.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0126
