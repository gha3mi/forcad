    subroutine forcad_nurbs_curve_0220(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        integer, parameter :: NFAMILY = 11
        real(rk), parameter :: NURBS_FIT_TOL = 1.0e-4_rk
        character(len=32), parameter :: family_name(NFAMILY) = [character(len=32) :: &
            "uniform unclamped",&
            "nonuniform unclamped A",&
            "uniform clamped",&
            "nonuniform clamped",&
            "uniform left clamped",&
            "uniform right clamped",&
            "nonuniform left clamped",&
            "nonuniform right clamped",&
            "uniform partially clamped",&
            "nonuniform partially clamped",&
            "nonuniform unclamped B"]
        real(rk), parameter :: knot_family(9,NFAMILY) = reshape([&
            0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 5.0_rk, 6.0_rk, 7.0_rk, 8.0_rk,&
            -1.4_rk, 0.0_rk, 0.3_rk, 1.1_rk, 2.6_rk, 3.0_rk, 4.4_rk, 4.7_rk, 5.5_rk,&
            0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 4.0_rk, 4.0_rk,&
            0.0_rk, 0.0_rk, 0.0_rk, 0.4_rk, 1.3_rk, 1.7_rk, 2.5_rk, 2.5_rk, 2.5_rk,&
            0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 5.0_rk, 6.0_rk,&
            -2.0_rk,-1.0_rk, 0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 4.0_rk, 4.0_rk,&
            0.0_rk, 0.0_rk, 0.0_rk, 0.2_rk, 1.4_rk, 2.0_rk, 3.5_rk, 4.1_rk, 5.0_rk,&
            -2.0_rk,-1.0_rk, 0.0_rk, 0.2_rk, 1.4_rk, 2.0_rk, 3.5_rk, 3.5_rk, 3.5_rk,&
            0.0_rk, 0.0_rk, 1.0_rk, 2.0_rk, 3.0_rk, 4.0_rk, 5.0_rk, 6.0_rk, 6.0_rk,&
            0.0_rk, 0.0_rk, 0.4_rk, 1.3_rk, 1.7_rk, 2.5_rk, 3.1_rk, 3.1_rk, 4.4_rk,&
            0.0_rk, 0.2_rk, 0.7_rk, 1.6_rk, 2.2_rk, 3.8_rk, 4.0_rk, 5.2_rk, 6.1_rk],&
            [9,NFAMILY])
        type(nurbs_curve) :: family_curve, refined_curve
        real(rk) :: family_knot(9), family_Xc(6,3), family_Wc(6), xi, dL, fit_error
        real(rk), allocatable :: active(:), Xt_family(:), Xg_family(:,:), Xg_reference(:,:)
        real(rk), allocatable :: T(:), dT(:), d2T(:), Tgrid(:,:), dT_dX(:,:), elevated_knot(:)
        integer, allocatable :: elem(:,:)
        integer :: family, point

        do point = 1, 6
            family_Xc(point,:) = [&
                real(point-1,rk),&
                0.2_rk*real((point-1)*(point-1),rk),&
                0.05_rk*real(point-1,rk)]
            family_Wc(point) = 1.0_rk + 0.03_rk*real(point-1,rk)
        end do

        do family = 1, NFAMILY
            family_knot = knot_family(:,family)
            active = active_knots(family_knot, 6, 2)
            xi = 0.5_rk*(active(1) + active(2))

            call family_curve%set(&
                knot   = family_knot,&
                Xc     = family_Xc,&
                degree = 2)
            call family_curve%create(res=9)
            Xt_family = family_curve%get_Xt()
            Xg_reference = family_curve%get_Xg()

            elem = family_curve%get_elem_Xg_vis()
            call family_curve%basis(res=3, Tgc=Tgrid)
            elem = family_curve%get_elem_Xg_vis()

            call family_curve%basis(xi, T)
            call family_curve%derivative(xi, dT, T)
            call family_curve%derivative2(xi, d2T, dT, T)

            elem = family_curve%cmp_elem()

            call family_curve%ansatz(&
                ie       = 1,&
                ig       = 1,&
                Tgc      = T,&
                dTgc_dXg = dT_dX,&
                dL       = dL,&
                ngauss   = 3)

            call refined_curve%set(&
                knot   = family_knot,&
                Xc     = family_Xc,&
                degree = 2)
            call refined_curve%insert_knots([xi], [1])
            call refined_curve%create(Xt=Xt_family)
            Xg_family = refined_curve%get_Xg()

            call refined_curve%remove_knots([xi], [1])
            call refined_curve%create(Xt=Xt_family)
            Xg_family = refined_curve%get_Xg()

            call refined_curve%set(&
                knot   = family_knot,&
                Xc     = family_Xc,&
                degree = 2)
            call refined_curve%elevate_degree(1)
            elevated_knot = refined_curve%get_knot()
            call refined_curve%create(Xt=Xt_family)
            Xg_family = refined_curve%get_Xg()

            call refined_curve%set(&
                knot   = family_knot,&
                Xc     = family_Xc,&
                degree = 2)
            call refined_curve%lsq_fit_bspline(Xt_family, Xg_reference, size(Xt_family))
            call refined_curve%create(Xt=Xt_family)
            Xg_family = refined_curve%get_Xg()
            fit_error = norm2(Xg_family-Xg_reference)/max(1.0_rk,norm2(Xg_reference))

            call family_curve%set(&
                knot   = family_knot,&
                Xc     = family_Xc,&
                Wc     = family_Wc,&
                degree = 2)
            call family_curve%create(Xt=Xt_family)
            Xg_reference = family_curve%get_Xg()
            call refined_curve%finalize()
            call refined_curve%set(&
                knot   = family_knot,&
                Xc     = family_Xc,&
                degree = 2)
            call refined_curve%lsq_fit_nurbs(&
                Xt    = Xt_family,&
                Xdata = Xg_reference,&
                ndata = size(Xt_family))
            call refined_curve%create(Xt=Xt_family)
            Xg_family = refined_curve%get_Xg()
            fit_error = norm2(Xg_family-Xg_reference)/max(1.0_rk,norm2(Xg_reference))

            call ut%test(ti)%check(&
                name     = trim(family_name(family))//": NURBS fit",&
                res      = fit_error,&
                expected = 0.0_rk,&
                tol      = NURBS_FIT_TOL,&
                msg      = "Nonlinear fit must reproduce compatible rational data.",&
                group    = "forcad_nurbs_curve")
            ti = ti + 1
        end do

    end subroutine forcad_nurbs_curve_0220
