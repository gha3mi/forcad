    subroutine forcad_nurbs_curve_0151(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: nurbs_curve_tol = 1.0e-5_rk
        type(nurbs_curve) :: line
        real(rk) :: Xline(2,3)

        Xline(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xline(2,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        call line%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xline)
        call line%put_to_nurbs(line%get_Xc(), reshape([1, 2], [1,2]))

        call ut%test(ti)%check(&
            name     = "put_to_nurbs(): curve maps control line",&
            res      = maxval(abs(line%get_Xg() - line%get_Xc())),&
            expected = 0.0_rk,&
            tol      = nurbs_curve_tol,&
            msg      = "Curve put_to_nurbs should map endpoint line exactly",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0151
