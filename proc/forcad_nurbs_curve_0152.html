    subroutine forcad_nurbs_curve_0152(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: cgen
        real(rk) :: Xgen(4,3)
        character(len=*), parameter :: fgen = "vtk/forcad_curve_general.iges"
        character(len=512) :: line
        integer :: unit, ios
        logical :: found

        Xgen(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xgen(2,:) = [1.0_rk, 0.5_rk, 0.0_rk]
        Xgen(3,:) = [2.0_rk, 0.5_rk, 0.0_rk]
        Xgen(4,:) = [3.0_rk, 0.0_rk, 0.0_rk]
        call cgen%set(2, 4, Xgen)
        call cgen%export_iges(fgen)
        found = .false.
        open(newunit=unit, file=fgen, status="old", action="read", iostat=ios)
        if (ios == 0) then
            do
                read(unit, "(A)", iostat=ios) line
                if (ios /= 0) exit
                if (index(line, "126,3,2,0,0,1") > 0) then
                    found = .true.
                    exit
                end if
            end do
            close(unit)
        end if

        call ut%test(ti)%check(&
            name     = "export_iges(): curve uses nc-derived IGES K",&
            res      = found,&
            expected = .true.,&
            msg      = "Curve IGES entity 126 should write K=nc-1 and polynomial PROP3=1",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0152
