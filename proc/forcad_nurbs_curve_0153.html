    subroutine forcad_nurbs_curve_0153(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: cproj
        real(rk) :: Xproj(2,3), Xt_out, Xg_out(3)

        Xproj(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xproj(2,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        call cproj%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xproj)
        call cproj%nearest_point2([0.2_rk, 0.3_rk, 0.0_rk], 1.0e-12_rk, 0, Xt_out, Xg_out)

        call ut%test(ti)%check(&
            name     = "nearest_point2(): maxit zero returns bounded curve parameter",&
            res      = (Xt_out >= 0.0_rk .and. Xt_out <= 1.0_rk .and. all(ieee_is_finite(Xg_out))),&
            expected = .true.,&
            msg      = "Nearest_point2(): maxit zero returns bounded curve parameter.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0153
