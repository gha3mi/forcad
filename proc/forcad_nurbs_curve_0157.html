    subroutine forcad_nurbs_curve_0157(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        type(nurbs_curve) :: cbad
        type(nurbs_curve) :: cmod
        real(rk) :: Xbad(2,3)
        real(rk) :: Xnew(3,3)
        real(rk) :: Wbad(2)
        real(rk) :: Wbad3(3)
        real(rk) :: Wgood(2)

        Xbad(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xbad(2,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        Xnew(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xnew(2,:) = [1.0_rk, 0.5_rk, 0.0_rk]
        Xnew(3,:) = [2.0_rk, 0.0_rk, 0.0_rk]
        Wbad = [1.0_rk, 0.0_rk]
        Wbad3 = [1.0_rk, 0.0_rk, 1.0_rk]
        Wgood = 1.0_rk
        call cbad%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wbad)

        call cmod%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wgood)
        call cmod%modify_Wc(0.5_rk, 1)
        call cmod%modify_Wc(1.0_rk, 1)

        call ut%test(ti)%check(&
            name     = "modify_Wc(): restores uniform cache",&
            res      = cmod%is_rational(),&
            expected = .false.,&
            msg      = "Restoring uniform curve weights must clear the rational flag",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0157
