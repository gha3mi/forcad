    subroutine forcad_nurbs_curve_0160(ut, ti)
        type(unit_tests), intent(inout) :: ut
        integer, intent(inout) :: ti

        real(rk), parameter :: nurbs_curve_tol = 1.0e-5_rk
        type(nurbs_curve) :: cbad, cmod, cstate, catomic
        real(rk) :: Xbad(2,3), Xnew(3,3), Wbad(2), Wbad3(3), Wgood(2), Wstate(2)
        real(rk), allocatable :: Xkeep(:,:)
        logical :: rejected
        integer :: preserved

        Xbad(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xbad(2,:) = [1.0_rk, 0.0_rk, 0.0_rk]
        Xnew(1,:) = [0.0_rk, 0.0_rk, 0.0_rk]
        Xnew(2,:) = [1.0_rk, 0.5_rk, 0.0_rk]
        Xnew(3,:) = [2.0_rk, 0.0_rk, 0.0_rk]
        Wbad = [1.0_rk, 0.0_rk]
        Wbad3 = [1.0_rk, 0.0_rk, 1.0_rk]
        Wgood = 1.0_rk
        call cbad%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wbad)

        call cmod%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wgood)
        call cmod%modify_Wc(0.5_rk, 1)
        call cmod%modify_Wc(1.0_rk, 1)
        call cmod%modify_Wc(-1.0_rk, 1)

        Wstate = [1.0_rk, 0.5_rk]
        call cstate%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wstate)
        call cstate%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad)

        call catomic%set([0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk], Xbad, Wstate)
        call catomic%set([0.0_rk, 0.0_rk, 0.0_rk, 1.0_rk, 1.0_rk, 1.0_rk], Xnew, Wbad3)
        rejected = .not. catomic%err%ok
        call catomic%err%reset()
        preserved = 0
        if (rejected .and. catomic%get_nc() == 2 .and. catomic%get_degree() == 1) then
            Xkeep = catomic%get_Xc()
            if (size(Xkeep, 1) == size(Xbad, 1) .and. size(Xkeep, 2) == size(Xbad, 2)) then
                if (maxval(abs(Xkeep - Xbad)) <= nurbs_curve_tol .and. catomic%is_rational()) preserved = 1
            end if
        end if

        call ut%test(ti)%check(&
            name     = "set(): preserves curve state after invalid weights",&
            res      = preserved,&
            expected = 1,&
            msg      = "Invalid weights must not change the curve state.",&
            group    = "forcad_nurbs_curve")
        ti = ti + 1

    end subroutine forcad_nurbs_curve_0160
